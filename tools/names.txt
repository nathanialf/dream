; Dream: Land of Giants (SNES prototype) curated labels. Format:
;   <24-bit addr> <name>      code entry (traced as a subroutine)
;   data <addr> <name>        data label only
;   ram <addr> <name>         WRAM / direct page variable
; Addresses are HiROM ($C0:0000 = file offset 0). Bank $00/$80 mirrors of the
; high half resolve to the same file offset.

; ---- entry points (from the vector table at file 0xFFE4) ----
C08000 reset                  ; native RESET: clears WRAM, inits sound, jumps to main loop
C0A4D1 nmi                    ; NMI: saves regs, reads RDNMI, then jmp ($0000) -> handler pointer in RAM
C0A442 unused_vec             ; COP/BRK/ABORT/IRQ vectors all point here
data C00000 stale_build_image     ; low half of bank $C0: older build leftover, calls into live code at off-by-one addresses (e.g. $ABCB)

; ---- sound (bank $C1 high half = $81:8000) ----
C18000 spc_init               ; jsl target: IPL handshake, upload loader + driver from $C2:0000
C1805A spc_ipl_upload_loader  ; uploads 0x88 bytes from $C2:0000 to SPC $04D8 via IPL protocol
C1809F spc_upload_driver      ; src ptr $04/$06 = $C2:0088, streams blocks via custom loader
C1830A upload_inline_spc_block ; block header at [$04]: word dest, word count, then words (name: p4plus2 DKC2 bank_B5)
C18324 upload_spc_block        ; (name: p4plus2 DKC2 bank_B5)
C183CE spc_command            ; jsl target, A = command (0 at boot, 1/2 from reset)

; ---- core ----
C0A385 ppu_init               ; writes most $21xx regs, NMITIMEN, clears OAM/VRAM
C0A2DE read_joypads           ; JOYSER0/1 + JOY1..4
C0A4A8 set_bg_scroll          ; writes BGxHOFS/VOFS, returns with M=16
C0A500 clear_sprite_table     ; zeroes $0400..$041E, $94=$0200
C08152 wait_and_enable_nmi    ; INIDISP + NMITIMEN setup
C08BDB ppu_regs_default
data C0826A game_mode_table        ; jsr (table,X) with X = game_mode*2

data C0814C alt_mode_table
data C20000 spc_loader_image  ; 0x88-byte SPC700 loader uploaded by IPL
data C20088 spc_driver_blocks ; block list uploaded by spc_upload_driver

; ---- RAM ----

ram 0004 ptr_04               ; generic 24-bit pointer ($04/$06): SPC upload source, handler dispatch via jmp ($0004), data reads

ram 00A4 game_mode            ; index into game_mode_table
